   6: Expired keys are never returned by get(). A failed refresh does not
   7: extend the previous expiry time. Callers must provide trusted loading
   8: and transport logic before using this cache in an application.
   9: """
  10: 
  11: from __future__ import annotations
  12: 
  13: from copy import deepcopy
  14: from threading import RLock
  15: from time import monotonic
  16: from typing import Any, Callable
  17: 
  18: from app.entra_metadata_client import (
  19:     MetadataTrustError,
  20:     validate_jwks_document,
  21: )
  22: 
  23: 
  24: class JWKSCacheError(RuntimeError):
  25:     """Raised when trusted JWKS data cannot be loaded or validated."""
  26: 
  27: 
  28: class JWKSCache:
  29:     """Thread-safe in-memory JWKS cache with bounded TTL."""
  30: 
  31:     def __init__(
  32:         self,
  33:         loader: Callable[[], Any],
  34:         *,
  35:         ttl_seconds: float = 300.0,
  36:         clock: Callable[[], float] = monotonic,
  37:     ) -> None:
  38:         if not callable(loader):
  39:             raise ValueError("loader must be callable")
  40:         if not isinstance(ttl_seconds, (int, float)) or isinstance(
  41:             ttl_seconds, bool
  42:         ):
  43:             raise ValueError("ttl_seconds must be numeric")
  44:         if not 0 < ttl_seconds <= 3600:
  45:             raise ValueError("ttl_seconds must be greater than 0 and at most 3600")
  46:         if not callable(clock):
  47:             raise ValueError("clock must be callable")
  48: 
  49:         self._loader = loader
  50:         self._ttl_seconds = float(ttl_seconds)
  51:         self._clock = clock
  52:         self._lock = RLock()
  53:         self._document: dict[str, Any] | None = None
  54:         self._expires_at: float | None = None
  55: 
  56:     def _load_and_validate(self) -> dict[str, Any]:
  57:         try:
  58:             document = self._loader()
  59:             validated = validate_jwks_document(document)
  60:             # Never expose the loader's mutable object to the cache.
  61:             return deepcopy(validated)
  62:         except (MetadataTrustError, TypeError, ValueError) as exc:
  63:             raise JWKSCacheError("Loaded JWKS failed validation") from exc
  64:         except Exception as exc:
  65:             raise JWKSCacheError("JWKS loader failed") from exc
  66: 
  67:     def get(self) -> dict[str, Any]:
  68:         """Return a copy of cached keys, loading only when missing/expired."""
  69:         with self._lock:
  70:             now = self._clock()
  71: 
  72:             if (
  73:                 self._document is not None
  74:                 and self._expires_at is not None
  75:                 and now < self._expires_at
  76:             ):
  77:                 return deepcopy(self._document)
  78: 
  79:             # Do not return expired keys if the loader fails.
  80:             document = self._load_and_validate()
  81:             self._document = document
  82:             self._expires_at = self._clock() + self._ttl_seconds
  83:             return deepcopy(document)
  84: 
  85:     def refresh(self) -> dict[str, Any]:
  86:         """Force a fresh load and replace the cache only after validation."""
  87:         with self._lock:
  88:             document = self._load_and_validate()
  89:             self._document = document
  90:             self._expires_at = self._clock() + self._ttl_seconds
  91:             return deepcopy(document)
  92: 
  93:     def invalidate(self) -> None:
  94:         """Discard cached keys so the next get() must reload."""
  95:         with self._lock:
  96:             self._document = None
  97:             self._expires_at = None
  98: 
  99:     @property
 100:     def is_valid(self) -> bool:
 101:         """Whether a non-expired cache entry currently exists."""
 102:         with self._lock:
 103:             return (
 104:                 self._document is not None
 105:                 and self._expires_at is not None
 106:                 and self._clock() < self._expires_at
 107:             )

   1: """RBAC policy foundation for CIRT DFIR Professional.
   2: 
   3: This module authorizes an already authenticated principal.
   4: It does not authenticate users or validate OIDC tokens.
   5: Do not treat a caller-created Principal as proof of identity.

   7: 
   8: from dataclasses import dataclass

  10: 
  11:

  14: ROLE_VIEWER = "Viewer"
  15: 
  16: KNOWN_ROLES: FrozenSet[str] = frozenset({
  17:     ROLE_ADMINISTRATOR,
  18:     ROLE_ANALYST,
  19:     ROLE_VIEWER,
  20: })
  21: 
  22: PERMISSIONS = {
  23:     "cases:read": frozenset({
  24:         ROLE_ADMINISTRATOR,
  25:         ROLE_ANALYST,
  26:         ROLE_VIEWER,
  27:     }),
  28:     "cases:update": frozenset({
  29:         ROLE_ADMINISTRATOR,
  30:         ROLE_ANALYST,
  31:     }),
  32:     "users:manage": frozenset({
  33:         ROLE_ADMINISTRATOR,
  34:     }),
  35:     "audit:read": frozenset({
  36:         ROLE_ADMINISTRATOR,
  37:     }),
  38: }
  39: 
  40: 
  41: @dataclass(frozen=True)
  42: class Principal:
  43:     """Identity and roles supplied by a trusted authentication layer."""
  44:     subject: str
  45:     roles: FrozenSet[str]
  46: 
  47: 
  48: class AuthenticationRequired(PermissionError):
  49:     """Raised when no authenticated principal is available."""
  50: 
  51: 
  52: class AuthorizationDenied(PermissionError):
  53:     """Raised when the principal lacks the required permission."""
  54: 
  55: 
  56: def validate_principal(principal: Principal | None) -> Principal:
  57:     if principal is None:
  58:         raise AuthenticationRequired("Authentication is required.")
  59:     if not isinstance(principal, Principal):
  60:         raise AuthenticationRequired("A trusted principal is required.")
  61:     if not isinstance(principal.subject, str) or not principal.subject.strip():
  62:         raise AuthenticationRequired("Principal subject is missing.")
  63:     if not principal.roles:
  64:         raise AuthorizationDenied("Principal has no assigned roles.")
  65:     if not principal.roles.issubset(KNOWN_ROLES):
  66:         raise AuthorizationDenied("Principal contains an unknown role.")
  67:     return principal
  68: 
  69: 
  70: def authorize(principal: Principal | None, permission: str) -> bool:
  71:     """Authorize a trusted principal for a named permission.
  72: 
  73:     The caller must obtain the principal from a verified authentication
  74:     mechanism, such as validated OIDC claims. Never build it from request
  75:     fields supplied by the client.
  76:     """
  77:     trusted = validate_principal(principal)
  78:     allowed_roles = PERMISSIONS.get(permission)
  79:     if allowed_roles is None:
  80:         raise AuthorizationDenied("Unknown permission.")
  81:     if trusted.roles.isdisjoint(allowed_roles):
  82:         raise AuthorizationDenied("Permission denied.")
  83:     return True
