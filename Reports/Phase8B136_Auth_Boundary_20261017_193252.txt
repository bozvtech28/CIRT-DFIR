"""Fail-closed authentication gate pending approved OIDC integration.

This module intentionally authenticates nobody. Replace the placeholder
identity provider integration only after the organization chooses a provider.
Never trust roles supplied by a browser or request body.
"""
from fastapi import Depends, HTTPException

from app.security import (
    AuthenticationRequired,
    AuthorizationDenied,
    Principal,
    authorize,
)

def get_current_principal() -> Principal:
    """Reject all requests until real token/session validation is implemented."""
    raise HTTPException(
        status_code=401,
        detail="Authentication is not configured. Access is denied by default.",
        headers={"WWW-Authenticate": "Bearer"},
    )

def require_permission(permission: str):
    """Require a validated principal and enforce the RBAC permission."""
    def dependency(principal: Principal = Depends(get_current_principal)) -> Principal:
        try:
            authorize(principal, permission)
        except AuthenticationRequired as exc:
            raise HTTPException(
                status_code=401,
                detail="Authentication required.",
                headers={"WWW-Authenticate": "Bearer"},
            ) from exc
        except AuthorizationDenied as exc:
            raise HTTPException(
                status_code=403,
                detail="Insufficient permission.",
            ) from exc
        return principal
    return dependency

@app.patch("/api/v1/cases/{case_id}", response_model=DemoCase)
def update_case_endpoint(
    case_id: str,
    payload: CaseUpdateRequest,
    principal = Depends(require_permission("cases:update")),
) -> DemoCase:
    """Update a case; audit attribution uses the authenticated principal subject."""
    if payload.status is None and payload.severity is None:
        raise HTTPException(
            status_code=422,
            detail="Provide status, severity, or both.",
        )

    try:
        case_store.ensure_database()
        db = connect(case_store.DB_PATH)
        try:
            updated = update_case(
                db,
                case_id=case_id,
                actor_id=principal.subject,
                timestamp=__import__("datetime").datetime.now(
                    __import__("datetime").timezone.utc
                ).isoformat(),
                status=payload.status,
                severity=payload.severity,
            )
        finally:
            db.close()
    except ValueError as exc:
        message = str(exc)
        if message == "Update contains no case changes.":
            raise HTTPException(status_code=409, detail=message) from exc
        raise HTTPException(status_code=422, detail=message) from exc

    if updated is None:
        raise HTTPException(status_code=404, detail="Case not found.")

    return DemoCase(**updated)

"""RBAC policy foundation for CIRT DFIR Professional.

This module authorizes an already authenticated principal.
It does not authenticate users or validate OIDC tokens.
Do not treat a caller-created Principal as proof of identity.
"""

from dataclasses import dataclass
from typing import FrozenSet


ROLE_ADMINISTRATOR = "Administrator"
ROLE_ANALYST = "DFIR Analyst"
ROLE_VIEWER = "Viewer"

KNOWN_ROLES: FrozenSet[str] = frozenset({
    ROLE_ADMINISTRATOR,
    ROLE_ANALYST,
    ROLE_VIEWER,
})

PERMISSIONS = {
    "cases:read": frozenset({
        ROLE_ADMINISTRATOR,
        ROLE_ANALYST,
        ROLE_VIEWER,
    }),
    "cases:update": frozenset({
        ROLE_ADMINISTRATOR,
        ROLE_ANALYST,
    }),
    "users:manage": frozenset({
        ROLE_ADMINISTRATOR,
    }),
    "audit:read": frozenset({
        ROLE_ADMINISTRATOR,
    }),
}


@dataclass(frozen=True)
class Principal:
    """Identity and roles supplied by a trusted authentication layer."""
    subject: str
    roles: FrozenSet[str]


class AuthenticationRequired(PermissionError):
    """Raised when no authenticated principal is available."""


class AuthorizationDenied(PermissionError):
    """Raised when the principal lacks the required permission."""


def validate_principal(principal: Principal | None) -> Principal:
    if principal is None:
        raise AuthenticationRequired("Authentication is required.")
    if not isinstance(principal, Principal):
        raise AuthenticationRequired("A trusted principal is required.")
    if not isinstance(principal.subject, str) or not principal.subject.strip():
        raise AuthenticationRequired("Principal subject is missing.")
    if not principal.roles:
        raise AuthorizationDenied("Principal has no assigned roles.")
    if not principal.roles.issubset(KNOWN_ROLES):
        raise AuthorizationDenied("Principal contains an unknown role.")
    return principal


def authorize(principal: Principal | None, permission: str) -> bool:
    """Authorize a trusted principal for a named permission.

    The caller must obtain the principal from a verified authentication
    mechanism, such as validated OIDC claims. Never build it from request
    fields supplied by the client.
    """
    trusted = validate_principal(principal)
    allowed_roles = PERMISSIONS.get(permission)
    if allowed_roles is None:
        raise AuthorizationDenied("Unknown permission.")
    if trusted.roles.isdisjoint(allowed_roles):
        raise AuthorizationDenied("Permission denied.")
    return True

def make_client_factory(handler):
    return partial(httpx.Client, transport=httpx.MockTransport(handler))


def json_response(document, status_code=200, headers=None):
    response_headers = {"content-type": "application/json"}
    if headers:
        response_headers.update(headers)
    return httpx.Response(
        status_code,
        headers=response_headers,
        content=json.dumps(document).encode("utf-8"),
    )

import json

import jwt
import pytest
from cryptography.hazmat.primitives.asymmetric import rsa
from jwt import InvalidTokenError

from app.entra_jwks import build_jwks_resolver


def make_pair(kid):
    private_key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
    public_jwk = json.loads(jwt.algorithms.RSAAlgorithm.to_jwk(private_key.public_key()))
    public_jwk.update({"kid": kid, "use": "sig", "alg": "RS256"})
    return private_key, public_jwk


def make_token(private_key, kid="key-a", algorithm="RS256"):
    return jwt.encode(
        {"sub": "synthetic-user", "iat": 1800000000, "nbf": 1800000000, "exp": 4102444800},
        private_key,
        algorithm=algorithm,
        headers={"kid": kid},
    )


def test_selects_matching_key_from_jwks():
    private_key, public_jwk = make_pair("key-a")
    resolver = build_jwks_resolver({"keys": [public_jwk]})
    token = make_token(private_key, "key-a")
    selected = resolver(token)
    decoded = jwt.decode(
        token,
        selected,
        algorithms=["RS256"],
        options={"verify_exp": False, "verify_iat": False, "verify_nbf": False},
    )
    assert decoded["sub"] == "synthetic-user"


def test_selects_correct_key_when_multiple_keys_exist():
    private_a, jwk_a = make_pair("key-a")
    _, jwk_b = make_pair("key-b")
    resolver = build_jwks_resolver({"keys": [jwk_a, jwk_b]})
    assert resolver(make_token(private_a, "key-a")) is not None

import unittest

from app.security import (

    Principal,

class TestRBACPolicy(unittest.TestCase):
    def setUp(self):
        self.admin = Principal("synthetic-admin", frozenset({ROLE_ADMINISTRATOR}))
        self.analyst = Principal("synthetic-analyst", frozenset({ROLE_ANALYST}))
        self.viewer = Principal("synthetic-viewer", frozenset({ROLE_VIEWER}))

    def test_admin_can_read_update_manage_users_and_read_audit(self):
        for permission in ("cases:read", "cases:update", "users:manage", "audit:read"):
            with self.subTest(permission=permission):
                self.assertTrue(authorize(self.admin, permission))

    def test_analyst_can_read_and_update_cases(self):
        self.assertTrue(authorize(self.analyst, "cases:read"))
        self.assertTrue(authorize(self.analyst, "cases:update"))

    def test_analyst_cannot_manage_users_or_read_audit(self):
        for permission in ("users:manage", "audit:read"):
            with self.subTest(permission=permission):
                with self.assertRaises(AuthorizationDenied):
                    authorize(self.analyst, permission)

    def test_viewer_can_read_cases(self):
        self.assertTrue(authorize(self.viewer, "cases:read"))

    def test_viewer_cannot_update_cases(self):
        with self.assertRaises(AuthorizationDenied):
            authorize(self.viewer, "cases:update")

    def test_missing_principal_is_rejected(self):
        with self.assertRaises(AuthenticationRequired):
            authorize(None, "cases:read")

    def test_unknown_role_is_rejected(self):
        principal = Principal("synthetic-user", frozenset({"SuperUser"}))
        with self.assertRaises(AuthorizationDenied):
            authorize(principal, "cases:read")

    def test_unknown_permission_is_rejected(self):
        with self.assertRaises(AuthorizationDenied):
            authorize(self.admin, "database:destroy")

    def test_empty_subject_is_rejected(self):
        principal = Principal(" ", frozenset({ROLE_ADMINISTRATOR}))
        with self.assertRaises(AuthenticationRequired):
            authorize(principal, "cases:read")

    def test_no_roles_is_rejected(self):
        principal = Principal("synthetic-user", frozenset())
        with self.assertRaises(AuthorizationDenied):
            authorize(principal, "cases:read")


if __name__ == "__main__":
    unittest.main()
