import json
import sqlite3
from pathlib import Path
from typing import Any


def connect(database_path: str | Path) -> sqlite3.Connection:
    connection = sqlite3.connect(str(database_path))
    connection.row_factory = sqlite3.Row
    connection.execute("PRAGMA foreign_keys = ON")
    return connection


def initialize(database_path: str | Path, schema_path: str | Path) -> None:
    schema = Path(schema_path).read_text(encoding="utf-8")
    db = connect(database_path)
    try:
        db.executescript(schema)
        db.commit()
    finally:
        db.close()


def create_case(
    db: sqlite3.Connection,
    *,
    case_id: str,
    title: str,
    severity: str,
    status: str,
    actor_id: str,
    timestamp: str,
) -> dict[str, Any]:
    if not all(isinstance(value, str) and value.strip() for value in (
        case_id, title, severity, status, actor_id, timestamp
    )):
        raise ValueError("Case fields, actor, and timestamp must be non-empty strings.")

    details = json.dumps(
        {"synthetic": True, "action": "case_created"},
        sort_keys=True,
        separators=(",", ":"),
    )

    with db:
        db.execute(
            """INSERT INTO cases
               (case_id, title, severity, status, created_at, updated_at, created_by)
               VALUES (?, ?, ?, ?, ?, ?, ?)""",
            (case_id, title, severity, status, timestamp, timestamp, actor_id),
        )
        db.execute(
            """INSERT INTO audit_events
               (case_id, event_type, actor_id, event_time, details_json)
               VALUES (?, ?, ?, ?, ?)""",
            (case_id, "CASE_CREATED", actor_id, timestamp, details),
        )

    return get_case(db, case_id)


def get_case(db: sqlite3.Connection, case_id: str) -> dict[str, Any] | None:
    row = db.execute(
        """SELECT case_id, title, severity, status, created_at,
                  updated_at, created_by, version
           FROM cases WHERE case_id = ?""",
        (case_id,),
    ).fetchone()
    return dict(row) if row else None


def list_cases(db: sqlite3.Connection) -> list[dict[str, Any]]:
    rows = db.execute(
        """SELECT case_id, title, severity, status, created_at,
                  updated_at, created_by, version
           FROM cases ORDER BY created_at, case_id"""
    ).fetchall()
    return [dict(row) for row in rows]


def list_audit_events(
    db: sqlite3.Connection, case_id: str
) -> list[dict[str, Any]]:
    rows = db.execute(
        """SELECT event_id, case_id, event_type, actor_id,
                  event_time, details_json, previous_hash, event_hash
           FROM audit_events
           WHERE case_id = ?
           ORDER BY event_id""",
        (case_id,),
    ).fetchall()
    return [dict(row) for row in rows]

CREATE TABLE IF NOT EXISTS cases (
    case_id TEXT PRIMARY KEY NOT NULL,
    title TEXT NOT NULL CHECK (length(trim(title)) > 0),
    severity TEXT NOT NULL CHECK (severity IN ('Low', 'Medium', 'High', 'Critical')),
    status TEXT NOT NULL CHECK (
        status IN ('Triage', 'Investigating', 'Evidence review', 'Contained', 'Closed')
    ),
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    created_by TEXT NOT NULL,
    version INTEGER NOT NULL DEFAULT 1 CHECK (version >= 1)
);

CREATE TABLE IF NOT EXISTS audit_events (
    event_id INTEGER PRIMARY KEY AUTOINCREMENT,
    case_id TEXT NOT NULL,
    event_type TEXT NOT NULL CHECK (length(trim(event_type)) > 0),
    actor_id TEXT NOT NULL CHECK (length(trim(actor_id)) > 0),
    event_time TEXT NOT NULL,
    details_json TEXT NOT NULL CHECK (json_valid(details_json)),
    previous_hash TEXT,
    event_hash TEXT,
    FOREIGN KEY (case_id) REFERENCES cases(case_id) ON DELETE RESTRICT
);

CREATE INDEX IF NOT EXISTS idx_cases_status ON cases(status);
CREATE INDEX IF NOT EXISTS idx_cases_severity ON cases(severity);
CREATE INDEX IF NOT EXISTS idx_audit_case_time ON audit_events(case_id, event_time);

-- Basic database-level protection against ordinary audit-event edits/deletes.
-- This does not protect against a database administrator or direct file tampering.
CREATE TRIGGER IF NOT EXISTS audit_events_no_update
BEFORE UPDATE ON audit_events
BEGIN
    SELECT RAISE(ABORT, 'Audit events are append-only');
END;

CREATE TRIGGER IF NOT EXISTS audit_events_no_delete
BEFORE DELETE ON audit_events
BEGIN
    SELECT RAISE(ABORT, 'Audit events are append-only');
END;

import json
import sqlite3
import tempfile
import unittest
from pathlib import Path

from case_repository import (
    connect, create_case, get_case, initialize, list_audit_events, list_cases
)

BASE = Path(__file__).resolve().parent
SCHEMA = BASE / "phase6_schema.sql"


class RepositoryTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory()
        self.database = Path(self.temp.name) / "synthetic-test.sqlite"
        initialize(self.database, SCHEMA)
        self.db = connect(self.database)

    def tearDown(self):
        self.db.close()
        self.temp.cleanup()

    def make_case(self, case_id="DEMO-REPO-001"):
        return create_case(
            self.db,
            case_id=case_id,
            title="Synthetic repository test",
            severity="High",
            status="Investigating",
            actor_id="synthetic-test",
            timestamp="2026-01-01T00:00:00Z",
        )

    def test_create_and_retrieve_case(self):
        case = self.make_case()
        self.assertEqual(case["case_id"], "DEMO-REPO-001")
        self.assertEqual(get_case(self.db, "DEMO-REPO-001")["severity"], "High")

    def test_list_cases(self):
        self.make_case("DEMO-REPO-002")
        self.make_case("DEMO-REPO-001")
        self.assertEqual(
            [c["case_id"] for c in list_cases(self.db)],
            ["DEMO-REPO-001", "DEMO-REPO-002"],
        )

    def test_missing_case_returns_none(self):
        self.assertIsNone(get_case(self.db, "DEMO-MISSING"))

    def test_creation_records_audit_event(self):
        self.make_case()
        events = list_audit_events(self.db, "DEMO-REPO-001")
        self.assertEqual(len(events), 1)
        self.assertEqual(events[0]["event_type"], "CASE_CREATED")
        self.assertEqual(events[0]["actor_id"], "synthetic-test")
        self.assertTrue(json.loads(events[0]["details_json"])["synthetic"])

    def test_invalid_severity_is_rejected(self):
        with self.assertRaises(sqlite3.IntegrityError):
            create_case(
                self.db,
                case_id="DEMO-INVALID",
                title="Synthetic invalid test",
                severity="Extreme",
                status="Triage",
                actor_id="synthetic-test",
                timestamp="2026-01-01T00:00:00Z",
            )

    def test_empty_actor_is_rejected(self):
        with self.assertRaises(ValueError):
            create_case(
                self.db,
                case_id="DEMO-EMPTY-ACTOR",
                title="Synthetic test",
                severity="Low",
                status="Triage",
                actor_id="",
                timestamp="2026-01-01T00:00:00Z",
            )

    def test_duplicate_case_id_is_rejected(self):
        self.make_case()
        with self.assertRaises(sqlite3.IntegrityError):
            self.make_case()

    def test_audit_event_cannot_be_updated(self):
        self.make_case()
        with self.assertRaises(sqlite3.IntegrityError):
            self.db.execute(
                "UPDATE audit_events SET event_type='ALTERED' WHERE event_id=1"
            )

    def test_audit_event_cannot_be_deleted(self):
        self.make_case()
        with self.assertRaises(sqlite3.IntegrityError):
            self.db.execute("DELETE FROM audit_events WHERE event_id=1")


if __name__ == "__main__":
    unittest.main(verbosity=2)
